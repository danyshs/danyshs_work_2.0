## expenseTracker.py:
```ts
import  pandas  as  pd
from  tag  import  tag_descriptions

file_path  =  r'.\cibc_processed.csv'
column_names  = ['Date', 'Description', 'Spent', 'Received']
df  =  pd.read_csv(file_path, names=column_names)
df['Date'] =  pd.to_datetime(df['Date'], dayfirst=True)
df  =  tag_descriptions(df)
grouped_df  =  df.groupby('Tag')[['Spent', 'Received']].sum()
total_sums  =  df[['Spent', 'Received']].sum()
total_sums_df  =  pd.DataFrame([total_sums])
grouped_df  =  pd.concat([grouped_df, total_sums_df])
grouped_df  =  grouped_df.rename(index={grouped_df.index[-1]: 'Total'})
print(grouped_df)
```
